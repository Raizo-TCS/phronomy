def after(seconds, &callback)
  if seconds <= 0
    callback.call
    return
  end
  add(&Phronomy::Execution.after(seconds, &callback))
end

def cancellation(token, &callback)
  return unless token

  super
  remaining = token.remaining_monotonic_seconds
  after(remaining, &callback) if !token.cancelled? && remaining
end
