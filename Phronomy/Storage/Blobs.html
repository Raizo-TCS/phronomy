def put_if_absent(key:, bytes:, attributes:)
  @view.validate!
  blob = Entry::Blob.new(key: key, bytes: bytes, attributes: resource.validate_attributes(attributes))
  @view.perform(:put_blob, resource, blob: blob)
end
