def fetch_async(query: nil, cancellation_token: nil, timeout: nil)
  Phronomy::Runtime.instance.blocking_io.submit(
    timeout: timeout,
    cancellation_token: cancellation_token
  ) do
    fetch(query: query, cancellation_token: cancellation_token)
  end
