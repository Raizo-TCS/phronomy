def initialize(agent:, name:)
  super(name: name)
  @operation_lease = agent.send(:__acquire_operation!)
  @lifetime_mutex = Mutex.new
  @work = 0
  @finished = false
  @released = false
end

def track(task)
  @lifetime_mutex.synchronize { @work += 1 }
  task.on_physical_complete do
    @lifetime_mutex.synchronize { @work -= 1 }
    release_if_quiet
  end
  task
end
