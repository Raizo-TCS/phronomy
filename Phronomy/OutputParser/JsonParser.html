def parse(text)
  json_str = extract_json(text)
  JSON.parse(json_str, symbolize_names: true)
rescue JSON::ParserError => e
  raise Phronomy::ParseError, "Failed to parse JSON: #{e.message}\nInput: #{text}"
end
